# List of domains
$domains = @(
    "pool.ntp.org",
    "time.windows.com",
)

# Port and protocol configuration
$localPort = 123
$remotePort = 123
$protocol = "UDP"

# Function to create or update firewall rule
function Create-FirewallRule {
    param (
        [string]$displayName,
        [string]$ipAddress,
        [string]$direction
    )

    # Construct rule display name
    $ruleDisplayName = "$displayName ($ipAddress)"
    
    # Check if the rule already exists
    $existingRule = Get-NetFirewallRule -DisplayName $ruleDisplayName -ErrorAction SilentlyContinue

    if ($existingRule) {
        Write-Host "Rule for $ipAddress already exists. Updating rule..."
        # Remove existing rule (overwrite)
        Remove-NetFirewallRule -DisplayName $ruleDisplayName
    }

    # Create Inbound Rule
    New-NetFirewallRule -DisplayName "$ruleDisplayName (Inbound)" `
        -Protocol $protocol `
        -Direction Inbound `
        -Action Allow `
        -LocalPort $localPort `
        -RemotePort $remotePort `
        -Enabled True `
        -RemoteAddress $ipAddress

    # Create Outbound Rule
    New-NetFirewallRule -DisplayName "$ruleDisplayName (Outbound)" `
        -Protocol $protocol `
        -Direction Outbound `
        -Action Allow `
        -LocalPort $localPort `
        -RemotePort $remotePort `
        -Enabled True `
        -RemoteAddress $ipAddress
}

# Function to resolve IP addresses for a domain
function Resolve-Domain {
    param (
        [string]$domain
    )

    # Resolve both IPv4 and IPv6 addresses
    $ipv4Addresses = (Resolve-DnsName $domain -Type A | Select-Object -ExpandProperty IPAddress)
    $ipv6Addresses = (Resolve-DnsName $domain -Type AAAA | Select-Object -ExpandProperty IPAddress)

    # Create firewall rules for each resolved IP address
    foreach ($ipv4 in $ipv4Addresses) {
        Create-FirewallRule -displayName "NTP - $domain" -ipAddress $ipv4 -direction "Inbound"
        Create-FirewallRule -displayName "NTP - $domain" -ipAddress $ipv4 -direction "Outbound"
    }

    foreach ($ipv6 in $ipv6Addresses) {
        Create-FirewallRule -displayName "NTP - $domain" -ipAddress $ipv6 -direction "Inbound"
        Create-FirewallRule -displayName "NTP - $domain" -ipAddress $ipv6 -direction "Outbound"
    }
}

# Resolve domains and create/update rules
foreach ($domain in $domains) {
    Resolve-Domain -domain $domain
}

Write-Host "Firewall rules have been successfully created/updated."
